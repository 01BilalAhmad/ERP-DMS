    22→import { Input } from '@/components/ui/input'
    23→import { Label } from '@/components/ui/label'

    63→// ---------- local types ----------
    64→interface Company {
    65→  id: string
    66→  code: string
    67→  name: string
    68→  salesTaxRate: number
    69→  filerTaxRate: number
    70→  nonFilerTaxRate: number
    71→  furtherTaxRate: number
    72→  defaultCurrency: string
    73→}
    74→interface ShopLink {
    75→  id: string
    76→  companyId: string
    77→  creditLimit: number
    78→  outstandingBalance: number
    79→}
    80→interface Shop {
    81→  id: string
    82→  code: string
    83→  name: string
    84→  ownerName?: string
    85→  phone?: string
    86→  shopClass: string
    87→  taxType: string
    88→  companyLinks: ShopLink[]
    89→}
    90→interface Product {
    91→  id: string
    92→  code: string
    93→  name: string
    94→  unit: string
    95→  tradePrice: number
    96→  retailerPrice?: number | null
    97→  taxRate: number
    98→  packSize?: string | null
    99→  availableStock: number
   100→  schemes?: any[]
   101→}
   102→interface Currency {
   103→  id: string
   104→  code: string
   105→  name: string
   106→  symbol: string
   107→  rate: number
   108→  isDefault: boolean
   109→}
   110→
   111→// ---------- main component ----------
   112→export function OrderEntryModule() {
   113→  const { toast } = useToast()
   114→  const session = useSession()
   115→  const role = session.data?.role as string | undefined
   116→
   117→  const companiesQ = useCompanies()
   118→  const currenciesQ = useCurrencies()
   119→  const createOrderMut = useCreateOrder()
   120→
   121→  const [companyId, setCompanyId] = useState<string>('')
   122→  const [shopId, setShopId] = useState<string>('')
   123→  const [productSearch, setProductSearch] = useState<string>('')
   124→  const [cart, setCart] = useState<CartItem[]>([])
   125→  const [manualDiscount, setManualDiscount] = useState<number>(0)
   126→  const [currencyCode, setCurrencyCode] = useState<string>('PKR')
   127→  const [notes, setNotes] = useState<string>('')
   128→  const [mobileCartOpen, setMobileCartOpen] = useState(false)
   129→  const [submittedOrder, setSubmittedOrder] = useState<any | null>(null)
   130→
   131→  const shopsQ = useShops({ companyId })
   132→  const productsQ = useProducts({ companyId, q: productSearch })
   133→
   134→  // Reset dependent state when company changes
   135→  useEffect(() => {
   136→    setShopId('')
   137→    setCart([])
   138→    setManualDiscount(0)
   139→    setProductSearch('')
   140→    setSubmittedOrder(null)
   141→  }, [companyId])
   142→
   143→  // Reset cart state when shop changes
   144→  useEffect(() => {
   145→    setCart([])
   146→    setManualDiscount(0)
   147→    setSubmittedOrder(null)
   148→  }, [shopId])
   149→
   150→  // Auto-select first company if user only has one
   151→  useEffect(() => {
   152→    if (!companyId && companiesQ.data?.length === 1) {
   153→      setCompanyId(companiesQ.data[0].id)
   154→    }
   155→  }, [companiesQ.data, companyId])
   156→
   157→  const companies = (companiesQ.data || []) as Company[]
   158→  const currencies = (currenciesQ.data || []) as Currency[]
   159→  const company = companies.find((c) => c.id === companyId)
   160→  const shop = ((shopsQ.data || []) as Shop[]).find((s) => s.id === shopId)
   161→  const companyLink = shop?.companyLinks?.find((l) => l.companyId === companyId)
   162→  const currency = currencies.find((c) => c.code === currencyCode)
   163→
   164→  // Compute totals using shared helper (no recompute)
   165→  const totals = useMemo(() => {
   166→    if (!company || !shop || cart.length === 0) return null
   167→    return calculateOrderTotals(cart, {
   168→      manualDiscount: Number(manualDiscount) || 0,
   169→      salesTaxRate: company.salesTaxRate,
   170→      filerTaxRate: company.filerTaxRate,
   171→      nonFilerTaxRate: company.nonFilerTaxRate,
   172→      furtherTaxRate: company.furtherTaxRate,
   173→      shopTaxType: shop.taxType as TaxType,
   174→      creditLimit: companyLink?.creditLimit || 0,
   175→      outstandingBalance: companyLink?.outstandingBalance || 0,
   176→      currencyRate: currency?.rate || 1,
   177→    })
   178→  }, [cart, manualDiscount, company, shop, companyLink, currency])
   179→
   180→  // ---------- cart operations ----------
   181→  const addToCart = useCallback((product: Product) => {
   182→    setCart((prev) => {
   183→      const ex = prev.find((i) => i.productId === product.id)
   184→      if (ex) {
   185→        return prev.map((i) =>
   186→          i.productId === product.id ? { ...i, quantity: i.quantity + 1 } : i
   187→        )
   188→      }
   189→      return [
   190→        ...prev,
   191→        {
   192→          productId: product.id,
   193→          code: product.code,
   194→          name: product.name,
   195→          unit: product.unit,
   196→          quantity: 1,
   197→          unitPrice: product.tradePrice,
   198→          discountPct: 0,
   199→          taxRate: product.taxRate,
   200→        },
   201→      ]
   202→    })
   203→    setSubmittedOrder(null)
   204→  }, [])
   205→
   206→  const updateItem = useCallback(
   207→    (productId: string, patch: Partial<CartItem>) => {
   208→      setCart((prev) =>
   209→        prev.map((i) => (i.productId === productId ? { ...i, ...patch } : i))
   210→      )
   211→    },
   212→    []
   213→  )
   214→
   215→  const removeItem = useCallback((productId: string) => {
   216→    setCart((prev) => prev.filter((i) => i.productId !== productId))
   217→  }, [])
   218→
   219→  const resetAll = useCallback(() => {
   220→    setCart([])
   221→    setManualDiscount(0)
   222→    setNotes('')
   223→    setSubmittedOrder(null)
   224→  }, [])
   225→
   226→  // ---------- submit ----------
   227→  const handleSubmit = async () => {
   228→    if (!company || !shop || cart.length === 0) {
   229→      toast({
   230→        title: 'Cannot submit',
   231→        description: 'Select company, shop and add at least one item.',
   232→        variant: 'destructive',
   233→      })
   234→      return
   235→    }
   236→    try {
   237→      const result: any = await createOrderMut.mutateAsync({
   238→        companyId,
   239→        shopId,
   240→        items: cart,
   241→        manualDiscount: Number(manualDiscount) || 0,
   242→        currency: currencyCode,
   243→        currencyRate: currency?.rate || 1,
   244→        notes: notes.trim() || undefined,
   245→        bookerId: session.data?.booker?.id,
   246→      })
   247→      setSubmittedOrder(result.order)
   248→      const warnings: string[] = result.warnings || []
   249→      toast({
   250→        title: `Order ${result.order.orderNo} created`,
   251→        description: warnings.length
   252→          ? `Grand total ${formatCurrency(result.order.grandTotal, result.order.currency, result.order.currencyRate)}. ${warnings.length} warning(s) flagged.`
   253→          : `Grand total ${formatCurrency(result.order.grandTotal, result.order.currency, result.order.currencyRate)}. Order is now pending approval.`,
   254→      })
   255→      setCart([])
   256→      setManualDiscount(0)
   257→      setNotes('')
   258→      setMobileCartOpen(false)
   259→    } catch (e: any) {
   260→      toast({
   261→        title: 'Failed to create order',
   262→        description: e.message || 'Unexpected error',
   263→        variant: 'destructive',
   264→      })
   265→    }
   266→  }
   267→
   268→  const cartCount = cart.length
   269→  const submitting = createOrderMut.isPending
   270→
   271→  // ---------- derived UI bits ----------
   272→  const filteredProducts = (productsQ.data || []) as Product[]
   273→
   274→  return (
   275→    <div className="space-y-5">
   276→      <PageHeader
   277→        title="Order Entry"
   278→        subtitle="Book a new sales order with full tax computation & warnings"
   279→        actions={
   280→          <div className="flex items-center gap-2">
   281→            {cartCount > 0 && (
   282→              <Button
   283→                variant="outline"
   284→                size="sm"
   285→                onClick={resetAll}
   286→                className="border-zinc-300 dark:border-zinc-700"
   287→              >
   288→                <RotateCcw className="w-4 h-4 mr-1" /> Reset
   289→              </Button>
   290→            )}
   291→            {role && !['ORDER_BOOKER', 'VIEWER'].includes(role) && (
   292→              <Badge
   293→                variant="outline"
   294→                className="hidden sm:inline-flex bg-emerald-50 text-emerald-700 border-emerald-200 dark:bg-emerald-950/40 dark:text-emerald-300 dark:border-emerald-900"
   295→              >
   296→                Role: {role.replace('_', ' ')}
   297→              </Badge>
   298→            )}
   299→          </div>
   300→        }
   301→      />
   302→
   303→      {/* Stepper */}
   304→      <div className="flex items-center gap-1 sm:gap-2 text-xs sm:text-sm overflow-x-auto pb-1">
   305→        <StepChip
   306→          num={1}
   307→          label="Company"
   308→          active={!companyId}
   309→          done={!!companyId}
   310→          value={company ? company.code : undefined}
   311→        />
   312→        <ChevronRight className="w-3.5 h-3.5 text-muted-foreground shrink-0" />
   313→        <StepChip
   314→          num={2}
   315→          label="Shop"
   316→          active={!!companyId && !shopId}
   317→          done={!!shopId}
   318→          value={shop ? shop.code : undefined}
   319→        />
   320→        <ChevronRight className="w-3.5 h-3.5 text-muted-foreground shrink-0" />
   321→        <StepChip
   322→          num={3}
   323→          label="Products"
   324→          active={!!shopId && cartCount === 0}
   325→          done={cartCount > 0}
   326→          value={cartCount > 0 ? `${cartCount} items` : undefined}
   327→        />
   328→        <ChevronRight className="w-3.5 h-3.5 text-muted-foreground shrink-0" />
   329→        <StepChip num={4} label="Submit" active={cartCount > 0} done={false} />
   330→      </div>
   331→
   332→      {/* Success banner */}
   333→      {submittedOrder && (
   334→        <Card className="border-emerald-300 bg-emerald-50 dark:border-emerald-900 dark:bg-emerald-950/30">
   335→          <CardContent className="p-4 flex items-start gap-3">
   336→            <CheckCircle2 className="w-5 h-5 text-emerald-600 dark:text-emerald-400 shrink-0 mt-0.5" />
   337→            <div className="flex-1 min-w-0">
   338→              <p className="font-semibold text-emerald-800 dark:text-emerald-200">
   339→                Order {submittedOrder.orderNo} created successfully
   340→              </p>
   341→              <p className="text-sm text-emerald-700 dark:text-emerald-300 mt-0.5">
   342→                Shop: {submittedOrder.shop?.name} · Grand total{' '}
   343→                {formatCurrency(
   344→                  submittedOrder.grandTotal,
   345→                  submittedOrder.currency,
   346→                  submittedOrder.currencyRate
   347→                )}{' '}
   348→                · Status: <span className="font-semibold">{submittedOrder.status}</span>
   349→              </p>
   350→              {submittedOrder.creditLimitExceeded || submittedOrder.stockShortage ? (
   351→                <p className="text-xs text-amber-700 dark:text-amber-400 mt-1 flex items-center gap-1">
   352→                  <AlertTriangle className="w-3 h-3" /> Order was flagged with warnings (credit / stock). Still saved successfully.
   353→                </p>
   354→              ) : null}
   355→            </div>
   356→            <Button
   357→              variant="ghost"
   358→              size="sm"
   359→              className="text-emerald-700 dark:text-emerald-300 hover:bg-emerald-100 dark:hover:bg-emerald-900/40"
   360→              onClick={() => setSubmittedOrder(null)}
   361→            >
   362→              Dismiss
   363→            </Button>
   364→          </CardContent>
   365→        </Card>
   366→      )}
   367→
   368→      {/* Main grid: left = selection + products, right = cart */}
   369→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   370→        <div className="lg:col-span-2 space-y-4">
   371→          {/* Step 1: Company */}
   372→          <Card className="border-zinc-200 dark:border-zinc-800">
   373→            <CardHeader className="pb-3">
   374→              <CardTitle className="text-sm font-semibold flex items-center gap-2">
   375→                <Building2 className="w-4 h-4 text-emerald-600 dark:text-emerald-400" />
   376→                1. Select Company
   377→              </CardTitle>
   378→            </CardHeader>
   379→            <CardContent>
   380→              {companiesQ.isLoading ? (
   381→                <Skeleton className="h-10 w-full" />
   382→              ) : companies.length === 0 ? (
   383→                <p className="text-sm text-muted-foreground">
   384→                  No companies assigned to you. Contact admin.
   385→                </p>
   386→              ) : (
   387→                <Select value={companyId} onValueChange={setCompanyId}>
   388→                  <SelectTrigger className="w-full">
   389→                    <SelectValue placeholder="Choose company…" />
   390→                  </SelectTrigger>
   391→                  <SelectContent>
   392→                    {companies.map((c) => (
   393→                      <SelectItem key={c.id} value={c.id}>
   394→                        <span className="font-mono text-xs mr-2">{c.code}</span>
   395→                        {c.name}
   396→                      </SelectItem>
   397→                    ))}
   398→                  </SelectContent>
   399→                </Select>
   400→              )}
   401→              {company && (
   402→                <div className="mt-3 grid grid-cols-2 sm:grid-cols-4 gap-2 text-xs">
   403→                  <TaxInfo label="Sales Tax" value={`${company.salesTaxRate}%`} />
   404→                  <TaxInfo label="Filer WHT" value={`${company.filerTaxRate}%`} />
   405→                  <TaxInfo label="Non-Filer WHT" value={`${company.nonFilerTaxRate}%`} />
   406→                  <TaxInfo label="Further Tax" value={`${company.furtherTaxRate}%`} />
   407→                </div>
   408→              )}
   409→            </CardContent>
   410→          </Card>
   411→
   412→          {/* Step 2: Shop */}
   413→          <Card className="border-zinc-200 dark:border-zinc-800">
   414→            <CardHeader className="pb-3">
   415→              <CardTitle className="text-sm font-semibold flex items-center gap-2">
   416→                <Store className="w-4 h-4 text-emerald-600 dark:text-emerald-400" />
   417→                2. Select Shop
   418→              </CardTitle>
   419→            </CardHeader>
   420→            <CardContent className="space-y-3">
   421→              {!companyId ? (
   422→                <p className="text-sm text-muted-foreground">
   423→                  Select a company first.
   424→                </p>
   425→              ) : shopsQ.isLoading ? (
   426→                <Skeleton className="h-10 w-full" />
   427→              ) : ((shopsQ.data || []) as Shop[]).length === 0 ? (
   428→                <p className="text-sm text-muted-foreground">
   429→                  No shops linked to this company.
   430→                </p>
   431→              ) : (
   432→                <>
   433→                  <Select value={shopId} onValueChange={setShopId}>
   434→                    <SelectTrigger className="w-full">
   435→                      <SelectValue placeholder="Choose shop…" />
   436→                    </SelectTrigger>
   437→                    <SelectContent>
   438→                      {((shopsQ.data || []) as Shop[]).map((s) => (
   439→                        <SelectItem key={s.id} value={s.id}>
   440→                          <span className="font-mono text-xs mr-2">{s.code}</span>
   441→                          {s.name}
   442→                          <span className="ml-2 text-[10px] text-muted-foreground">
   443→                            ({s.taxType})
   444→                          </span>
   445→                        </SelectItem>
   446→                      ))}
   447→                    </SelectContent>
   448→                  </Select>
   449→
   450→                  {shop && companyLink && (
   451→                    <div className="grid grid-cols-2 sm:grid-cols-4 gap-2 p-3 rounded-lg bg-emerald-50 dark:bg-emerald-950/30 border border-emerald-200 dark:border-emerald-900">
   452→                      <ShopInfoMini
   453→                        label="Tax Type"
   454→                        value={shop.taxType}
   455→                        tone={shop.taxType === 'FILER' ? 'emerald' : 'amber'}
   456→                      />
   457→                      <ShopInfoMini label="Class" value={shop.shopClass} />
   458→                      <ShopInfoMini
   459→                        label="Outstanding"
   460→                        value={formatCurrency(companyLink.outstandingBalance)}
   461→                      />
   462→                      <ShopInfoMini
   463→                        label="Credit Limit"
   464→                        value={
   465→                          companyLink.creditLimit > 0
   466→                            ? formatCurrency(companyLink.creditLimit)
   467→                            : 'Unlimited'
   468→                        }
   469→                      />
   470→                    </div>
   471→                  )}
   472→                </>
   473→              )}
   474→            </CardContent>
   475→          </Card>
   476→
   477→          {/* Step 3: Products */}
   478→          <Card className="border-zinc-200 dark:border-zinc-800">
   479→            <CardHeader className="pb-3">
   480→              <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-2">
   481→                <CardTitle className="text-sm font-semibold flex items-center gap-2">
   482→                  <Package className="w-4 h-4 text-emerald-600 dark:text-emerald-400" />
   483→                  3. Add Products
   484→                </CardTitle>
   485→                <div className="relative w-full sm:w-72">
   486→                  <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   487→                  <Input
   488→                    placeholder="Search by name or code…"
   489→                    value={productSearch}
   490→                    onChange={(e) => setProductSearch(e.target.value)}
   491→                    className="pl-8 h-9"
   492→                    disabled={!shopId}
   493→                  />
   494→                </div>
   495→              </div>
   496→            </CardHeader>
   497→            <CardContent>
   498→              {!shopId ? (
   499→                <p className="text-sm text-muted-foreground py-4 text-center">
   500→                  Select a shop to start adding products.
   501→                </p>
   502→              ) : productsQ.isLoading ? (
   503→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
   504→                  {Array.from({ length: 6 }).map((_, i) => (
   505→                    <Skeleton key={i} className="h-20 w-full" />
   506→                  ))}
   507→                </div>
   508→              ) : filteredProducts.length === 0 ? (
   509→                <EmptyState
   510→                  icon={Package}
   511→                  title="No products found"
   512→                  hint="Try a different search term, or add products for this company first."
   513→                />
   514→              ) : (
   515→                <ScrollArea className="max-h-[420px]">
   516→                  <div className="grid grid-cols-1 sm:grid-cols-2 gap-2 pr-1">
   517→                    {filteredProducts.slice(0, 60).map((p) => {
   518→                      const inCart = cart.find((i) => i.productId === p.id)
   519→                      const lowStock =
   520→                        p.availableStock !== undefined && p.availableStock <= 5
   521→                      return (
   522→                        <div
   523→                          key={p.id}
   524→                          className={cn(
   525→                            'group rounded-lg border p-3 transition-colors',
   526→                            inCart
   527→                              ? 'border-emerald-400 bg-emerald-50/60 dark:border-emerald-800 dark:bg-emerald-950/30'
   528→                              : 'border-zinc-200 dark:border-zinc-800 hover:border-emerald-300 dark:hover:border-emerald-800'
   529→                          )}
   530→                        >
   531→                          <div className="flex items-start justify-between gap-2">
   532→                            <div className="min-w-0 flex-1">
   533→                              <p className="text-sm font-medium truncate">
   534→                                {p.name}
   535→                              </p>
   536→                              <p className="text-[11px] text-muted-foreground font-mono">
   537→                                {p.code}
   538→                                {p.packSize ? ` · ${p.packSize}` : ''}
   539→                              </p>
   540→                            </div>
   541→                            {inCart ? (
   542→                              <Badge className="bg-emerald-100 text-emerald-700 dark:bg-emerald-950/50 dark:text-emerald-300 text-[10px]">
   543→                                In cart: {inCart.quantity}
   544→                              </Badge>
   545→                            ) : null}
   546→                          </div>
   547→                          <div className="flex items-center justify-between mt-2">
   548→                            <div className="text-xs space-y-0.5">
   549→                              <p className="font-semibold">
   550→                                {formatCurrency(p.tradePrice)}
   551→                                <span className="text-muted-foreground font-normal">
   552→                                  {' '}/ {p.unit}
   553→                                </span>
   554→                              </p>
   555→                              <p
   556→                                className={cn(
   557→                                  'flex items-center gap-1',
   558→                                  lowStock
   559→                                    ? 'text-amber-600 dark:text-amber-400'
   560→                                    : 'text-muted-foreground'
   561→                                )}
   562→                              >
   563→                                Stock: {p.availableStock ?? 0} {p.unit}
   564→                                {lowStock && (
   565→                                  <AlertTriangle className="w-3 h-3" />
   566→                                )}
   567→                              </p>
   568→                            </div>
   569→                            <Button
   570→                              size="sm"
   571→                              onClick={() => addToCart(p)}
   572→                              className="bg-emerald-600 hover:bg-emerald-700 text-white h-8"
   573→                            >
   574→                              <Plus className="w-3.5 h-3.5 mr-1" />

   751→// ---------- Cart panel ----------
   752→function CartPanel({
   753→  shop,
   754→  cart,
   755→  totals,
   756→  manualDiscount,
   757→  setManualDiscount,
   758→  currencyCode,
   759→  setCurrencyCode,
   760→  currencies,
   761→  notes,
   762→  setNotes,
   763→  updateItem,
   764→  removeItem,
   765→  onSubmit,
   766→  submitting,
   767→  currency,
   768→  company,
   769→  compact = false,
   770→}: {
   771→  shop?: Shop
   772→  cart: CartItem[]
   773→  totals: ReturnType<typeof calculateOrderTotals> | null
   774→  manualDiscount: number
   775→  setManualDiscount: (v: number) => void
   776→  currencyCode: string
   777→  setCurrencyCode: (v: string) => void
   778→  currencies: Currency[]
   779→  notes: string
   780→  setNotes: (v: string) => void
   781→  updateItem: (id: string, patch: Partial<CartItem>) => void
   782→  removeItem: (id: string) => void
   783→  onSubmit: () => void
   784→  submitting: boolean
   785→  currency?: Currency
   786→  company?: Company
   787→  compact?: boolean
   788→}) {

   790→    return (
   791→      <Card className="border-zinc-200 dark:border-zinc-800 sticky top-4">
   792→        <CardContent className="p-4">
   793→          <EmptyState
   794→            icon={ShoppingCart}
   795→            title="Cart is empty"
   796→            hint="Pick a shop first to start adding items."
   797→          />
   798→        </CardContent>
   799→      </Card>
   800→    )
   801→  }
   802→
   803→  return (
   804→    <Card className="border-emerald-200 dark:border-emerald-900">
   805→      <CardHeader className="pb-3 border-b border-zinc-100 dark:border-zinc-800">
   806→        <div className="flex items-center justify-between">
   807→          <CardTitle className="text-sm font-semibold flex items-center gap-2">
   808→            <ShoppingCart className="w-4 h-4 text-emerald-600 dark:text-emerald-400" />
   809→            Cart
   810→            {cart.length > 0 && (

   833→                  lineGross - (lineGross * item.discountPct) / 100

   871→                          type="number"

   879→                          }
   880→                          className="h-8 text-sm"

   958→                  <TotalsRow
   959→                    label="Manual Discount"
   960→                    value={`− ${formatCurrency(totals.manualDiscount, currencyCode, currency?.rate || 1)}`}
   961→                    tone="rose"
   962→                  />
   963→                )}
   964→                <TotalsRow
   965→                  label="Taxable Amount"
   966→                  value={formatCurrency(totals.taxableAmount, currencyCode, currency?.rate || 1)}
